use std::io::{self, Read, Write};
use std::process::Command;

fn enable_raw_mode() {
    // Sauvegarde les paramètres actuels du terminal
    Command::new("stty")
        .arg("-echo")
        .arg("raw")
        .status()
        .expect("Échec de l'activation du mode brut");
}

fn disable_raw_mode() {
    // Restaure les paramètres du terminal
    Command::new("stty")
        .arg("echo")
        .arg("-raw")
        .status()
        .expect("Échec de la désactivation du mode brut");
}

fn main() {
    println!("Choisir: 1- Pour Débit\n         2- Pour Crédit\n         3- Pour transfert");
    print!("Appuie sur 1, 2 ou 3 (sans Enter)... ou 0 pour annuler ");
    io::stdout().flush().unwrap();
    enable_raw_mode();

    let mut buffer = [0; 1]; // Lire un seul octet
    loop {
        io::stdin().read_exact(&mut buffer).unwrap();
        match buffer[0] {
            b'0' => {
                disable_raw_mode();
                println!("Ok, exit");
                break;
            }
            b'1' => {
                disable_raw_mode();
                println!("Débit");
                break;
            }
            b'2' => {
                disable_raw_mode();
                println!("Crédit");
                break;
            }
            b'3' => {
                disable_raw_mode();
                println!("Transfert");
                break;
            }
            _ => {
//                println!("\nTouche non reconnue : '{}'\r", buffer[0] as char);
            }
        }
    }
}
